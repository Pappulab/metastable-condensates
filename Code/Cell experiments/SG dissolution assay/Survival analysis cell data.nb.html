library(readr)
library(dplyr)
library(ggplot2)
library(tidyr)
library(survival)
library(survminer)
library(gridExtra)
df1 <- read_csv("manual_scoring.csv", col_types = "ffddd")
df2 <- read_csv("manual_scoring_negControl.csv", col_types = "ffddd")
df1 <- df1 %>% mutate(cat = "expressing")
df2 <- df2 %>% mutate(cat = "nonexpressing")
df3 <- bind_rows(df1, df2)
df4 <- df3 %>% filter(!((Day == 20231005 & Variant == "WT")|
                 (Day == 20231005 & Variant == "D262V")|
                 (Day == 20231207 & Variant == "allW D262V")))

df4 <- df4 %>% mutate(Index = interaction(Day, Variant))
df4$Index <- droplevels(df4$Index)
df4 <- df4 %>% mutate(t = t_dissolution - 62)

indices <- levels(df4$Index)
variants <- levels(df4$Variant)

df4.1 <- df4 %>% filter(cat == "expressing")
df4.2 <- df4 %>% filter(cat == "nonexpressing")
test1 <- df4.1 %>% mutate(status = case_when(t_dissolution == 181 ~ 0, TRUE ~ 1))
sfit1 <- survfit(Surv(t, status)~Variant, data=test1)
p1 <- ggsurvplot(sfit1, conf.int=TRUE, pval=TRUE, risk.table=F, 
           palette=c("dodgerblue2", "orchid2", "darkgreen"),
           legend.labs=c("WT", "D262V", "allW D262V"), legend.title="Variant",
           #legend = "bottom",
           xlab = "Recovery time (min)",
           ylab = "Fraction of cells with SGs",
           break.x.by = 15,
           pval.coord = c(85,1),
           risk.table.height=0.25,
           risk.table.title = "Number of cells with SGs",
           title = "SG disassembly in\neGFP-expressing cells",
           ggtheme = theme_classic() +
             theme(axis.title = element_text(size = 14,face = "bold"),
                   legend.text = element_text(size = 12),
                   legend.title = element_text(size =12, face = "bold"),
                   axis.text = element_text(size = 12),
                   title = element_text(size = 12, face = "bold")
                   ))

res1 <- pairwise_survdiff(Surv(t, status)~Variant, data=test1, p.adjust.method = "BH")
res1

    Pairwise comparisons using Log-Rank test 

data:  test1 and Variant 

           WT    D262V  
D262V      0.122 -      
allW D262V 0.036 7.5e-05

P value adjustment method: BH 
table1 <- res1$p.value
table1 <- signif(table1, digits = 2)
p1$plot <- p1$plot + annotation_custom(tableGrob(table1), xmin=0, xmax=75, ymin=0, ymax=0.25)
grob <- text_grob(label = "Adjusted p-values", size = 14, face = "bold")
p1$plot <- p1$plot + annotation_custom(grob, xmin = 0, xmax = 60, ymin = 0.3, ymax = 0.35)

pdf("surv_expressing.pdf", width = 4.75, height = 5)
print(p1)

dev.off()
